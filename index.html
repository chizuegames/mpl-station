function handleRoomClick(roomName) { const room = state.rooms[roomName]; if (!room) { return; } /* ===================================================== COMPROBAR SI EL CUARTO ES ADYACENTE A NUESTRA POSICIÓN ACTUAL ===================================================== */ const adjacentRooms = getAdjacentRooms(state.currentRoom); const canEnter = adjacentRooms.some( connection => connection.room === roomName ); /* Si el cuarto no está conectado con nuestra posición actual, no podemos entrar. */ if (!canEnter) { /* Si estamos tocando nuestra propia habitación actual, simplemente no hacemos nada. */ if (state.currentRoom === roomName) { showMessage( "ESTÁS EN
ESTA SALA" ); } return; } /* ===================================================== CUARTO MARCADO POR EL ESCÁNER PRIMER TOQUE: REVELAMOS SU CONTENIDO ===================================================== */ if ( room.status === "scannable" ) { room.status = "revealed"; removeMapIcon( `scan-${roomName}` ); addMapIcon( `enemy-${roomName}`, room.encounter, roomName, "enemy-icon" ); showMessage( `${room.label}
−2 ENERGÍAS` ); return; } /* ===================================================== CUARTO YA REVELADO SEGUNDO TOQUE: ENTRAMOS ===================================================== */ if ( room.status === "revealed" ) { /* A12 tiene su pantalla de encuentro propia. */ if ( roomName === "A12" ) { state.scannerActive = false; turnOffScanner(); openA12Encounter(); return; } /* Cuartos B. Por ahora simplemente nos movemos hasta ellos. */ room.status = "entered"; state.scannerActive = false; turnOffScanner(); moveToRoom( roomName ); showMessage( `ENTRASTE A
${roomName}` ); return; } /* ===================================================== CUARTO NO ESCANEADO EL ESCÁNER ES OPCIONAL. Podemos entrar directamente sin conocer previamente qué hay dentro. ===================================================== */ if ( room.status === "hidden" ) { /* ENTRADA DIRECTA A A12 */ if ( roomName === "A12" ) { state.scannerActive = false; turnOffScanner(); openA12Encounter(); return; } /* ENTRADA DIRECTA A UN CUARTO B */ room.status = "entered"; state.scannerActive = false; turnOffScanner(); moveToRoom( roomName ); showMessage( `ENTRASTE A
${roomName}` ); return; } /* ===================================================== SALA YA REVISADA ===================================================== */ if ( room.status === "completed" || room.status === "entered" ) { moveToRoom( roomName ); } }
